PHASE 1 — SETUP AND BUILD SIMPLESCALAR
===================================================================

# Create main project directory
mkdir Cache_Prefetching_Schemes
cd Cache_Prefetching_Schemes

# ----------------------------------
# Clone SimpleScalar 3.0 (Todd Austin)
# ----------------------------------
mkdir simplescalar
cd simplescalar

git clone https://github.com/toddmaustin/simplesim-3.0.git

# Enter simulator directory
cd simplesim-3.0

# Configure for PISA ISA
make config-pisa

# Build all simulator binaries
make             # or: make -j$(nproc)

# Return to project root
cd ../../

===================================================================
PHASE 4 — CACTI ENERGY EXTRACTION
===================================================================

# ---------------------------------------------------------------
# Clone CACTI and build
# ---------------------------------------------------------------

cd CACTI
git clone https://github.com/HewlettPackard/cacti.git
cd cacti
make
cd ../..

# ---------------------------------------------------------------
# Copy custom CACTI configuration files
# ---------------------------------------------------------------
# The il1.cfg, dl1.cfg, and l2.cfg files are provided in the
# project root directory (Cache_Prefetching_Schemes/).
# They MUST be copied into the CACTI/cacti folder before running.
# ---------------------------------------------------------------

cp il1.cfg CACTI/cacti/il1.cfg
cp dl1.cfg CACTI/cacti/dl1.cfg
cp l2.cfg CACTI/cacti/l2.cfg

# ---------------------------------------------------------------
# Run CACTI to generate modeled energy values
# ---------------------------------------------------------------

cd CACTI/cacti

./cacti -infile il1.cfg > il1.out
./cacti -infile dl1.cfg > dl1.out
./cacti -infile l2.cfg  > l2.out

cd ../..

===================================================================
PHASE 5 — MERGE CACTI ENERGY INTO FINAL CSV
===================================================================

python3 Memory_Modeling/scripts/merge_cacti_energy.py \
    results/summary_with_energy.csv \
    results/summary_with_energy_final.csv \
    CACTI/cacti/il1.out \
    CACTI/cacti/dl1.out \
    CACTI/cacti/l2.out
